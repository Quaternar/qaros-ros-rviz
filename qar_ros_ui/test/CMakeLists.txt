# Unit tests for qar_ros_ui.
#
# GoogleTest comes from the top-level qar_require_googletest(); this directory just uses
# its targets.
#
# Most tests need no ROS graph at all: they load the distro's typesupport DLLs
# and exercise introspection, conversion and serialization over real message
# types. A few create a node, a typed publisher/server and a bridge in-process
# (own domain id, bounded waits, no daemon) — see test_support.hpp. Those talk
# to typed entities from test_msgs / geometry_msgs / std_srvs; without them the
# live tests compile out.
#
# Running from a plain shell: the exe imports rclcpp.dll and friends, so
# <distro>/bin must be on PATH BEFORE gtest enumerates the tests.
# gtest_discover_tests' PRE_TEST mode enumerates inside the ctest process, and
# ctest include()s every file in the directory's TEST_INCLUDE_FILES in order —
# so a tiny script that prepends PATH, appended to that property before the
# discovery file, sets the environment for both the enumeration and the runs.
# (Build-time discovery would need the ROS env in the *build* shell instead.)

find_package(test_msgs QUIET)
find_package(geometry_msgs QUIET)
find_package(std_srvs QUIET)

add_executable(qar_ros_ui_tests
  test_main.cpp
  type_support_registry_tests.cpp
  introspection_tests.cpp
  dynamic_message_tests.cpp
  qos_selection_tests.cpp
  graph_discovery_tests.cpp
  topic_bridge_tests.cpp
  service_bridge_tests.cpp
  action_bridge_tests.cpp
)

target_link_libraries(qar_ros_ui_tests PRIVATE
  qar_rviz_plugin::qar_ros_ui
  GTest::gtest
)

if(test_msgs_FOUND AND geometry_msgs_FOUND AND std_srvs_FOUND)
  target_link_libraries(qar_ros_ui_tests PRIVATE
    test_msgs::test_msgs__rosidl_typesupport_cpp
    geometry_msgs::geometry_msgs__rosidl_typesupport_cpp
    std_srvs::std_srvs__rosidl_typesupport_cpp
  )
  target_compile_definitions(qar_ros_ui_tests PRIVATE QAR_ROS_UI_LIVE_TESTS=1)
else()
  message(WARNING
    "test_msgs / geometry_msgs / std_srvs not all found; the live qar_ros_ui tests are compiled out.")
endif()

target_compile_definitions(qar_ros_ui_tests PRIVATE
  "QAR_ROS_UI_TEST_ROS_ROOT=\"${QAR_ROS2_INSTALL_ROOT}\""
)

target_compile_features(qar_ros_ui_tests PRIVATE cxx_std_17)

qar_apply_warning_flags(qar_ros_ui_tests)

# --- Test-time PATH (see the block comment at the top) ----------------------
set(_qar_ros_bins "${QAR_ROS2_INSTALL_ROOT}/bin")
file(GLOB _qar_ros_opt_bins LIST_DIRECTORIES true "${QAR_ROS2_INSTALL_ROOT}/opt/*/bin")
foreach(_dir IN LISTS _qar_ros_opt_bins)
  if(IS_DIRECTORY "${_dir}")
    list(APPEND _qar_ros_bins "${_dir}")
  endif()
endforeach()
if(WIN32)
  set(_qar_path_sep ";")
else()
  set(_qar_path_sep ":")
endif()
string(REPLACE ";" "${_qar_path_sep}" _qar_ros_path_prepend "${_qar_ros_bins}")

set(_qar_ros_env_script "${CMAKE_CURRENT_BINARY_DIR}/qar_ros_ui_test_env.cmake")
file(WRITE "${_qar_ros_env_script}"
  "# Generated by qar_ros_ui/test/CMakeLists.txt: ROS runtime DLLs for test discovery and runs.\n"
  "set(ENV{PATH} \"${_qar_ros_path_prepend}${_qar_path_sep}\$ENV{PATH}\")\n"
)
set_property(DIRECTORY APPEND PROPERTY TEST_INCLUDE_FILES "${_qar_ros_env_script}")

file(TO_NATIVE_PATH "${QAR_ROS2_INSTALL_ROOT}/bin" _qar_ros_bin_native)

include(GoogleTest)
gtest_discover_tests(qar_ros_ui_tests
  DISCOVERY_MODE PRE_TEST
  DISCOVERY_TIMEOUT 120
  PROPERTIES
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_qar_ros_bin_native}"
    # The tests exercise the bridges, not the transport. Fast DDS's shared-memory transport is off for
    # these short-lived processes: participant teardown crashes intermittently and leaves stale segments.
    ENVIRONMENT "FASTDDS_BUILTIN_TRANSPORTS=UDPv4"
    TIMEOUT 120
    # Live DDS tests: never two at once under `ctest -j` (they share the host's
    # DDS transports; GraphDiscoveryLive.* crashed once in three -j8 runs).
    RESOURCE_LOCK qar_ros_dds
)
